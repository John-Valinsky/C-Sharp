# 14. Write a program to demonstrate one dimensional indexer. 

using System;

class NumberArray
{
    private int[] arr = new int[5];

    // 🔹 Indexer
    public int this[int index]
    {
        get
        {
            return arr[index];
        }
        set
        {
            arr[index] = value;
        }
    }
}

class Program
{
    static void Main()
    {
        NumberArray obj = new NumberArray();

        // Assign values using indexer
        for (int i = 0; i < 5; i++)
        {
            obj[i] = (i + 1) * 10;
        }

        // Display values using indexer
        Console.WriteLine("Array Elements:");
        for (int i = 0; i < 5; i++)
        {
            Console.WriteLine("Index " + i + ": " + obj[i]);
        }

        Console.ReadLine();
    }
}
