# 15. Write a program to demonstrate difference between static and instance constructors.  

using System;

class Demo
{
    static int count;
    int id;

    static Demo()
    {
        count = 100;
        Console.WriteLine("Static Constructor Called");
    }

    public Demo(int i)
    {
        id = i;
        Console.WriteLine("Instance Constructor Called for ID: " + id);
    }

    public void Display()
    {
        Console.WriteLine("ID: " + id + ", Count: " + count);
    }
}

class Program
{
    static void Main()
    {
        Demo dl = new Demo(1);
        dl.Display();

        Demo d2 = new Demo(2);
        d2.Display();
    }
}
