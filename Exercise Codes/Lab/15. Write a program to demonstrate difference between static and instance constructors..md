# 15. Write a program to demonstrate difference between static and instance constructors.  

using System;

class Demo
{
    static int count;
    int id;

    // 🔹 Static Constructor
    static Demo()
    {
        count = 100;
        Console.WriteLine("Static Constructor Called");
    }

    // 🔹 Instance Constructor
    public Demo(int i)
    {
        id = i;
        Console.WriteLine("Instance Constructor Called for ID: " + id);
    }

    public void Display()
    {
        Console.WriteLine("ID: " + id + ", Count: " + count);
    }
}

class Program
{
    static void Main()
    {
        Demo d1 = new Demo(1);
        d1.Display();

        Demo d2 = new Demo(2);
        d2.Display();

        Console.ReadLine();
    }
}
