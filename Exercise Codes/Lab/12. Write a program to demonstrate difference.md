# 11. Write a program to demonstrate method overriding.  

using System;

class Animal
{
    // Base class method
    public virtual void Sound()
    {
        Console.WriteLine("Animal makes a sound");
    }
}

class Dog : Animal
{
    // Overriding method
    public override void Sound()
    {
        Console.WriteLine("Dog barks");
    }
}

class Cat : Animal
{
    // Overriding method
    public override void Sound()
    {
        Console.WriteLine("Cat meows");
    }
}

class Program
{
    static void Main()
    {
        Animal a;

        a = new Dog();
        a.Sound();   // Calls Dog's method

        a = new Cat();
        a.Sound();   // Calls Cat's method

        Console.ReadLine();
    }
}
