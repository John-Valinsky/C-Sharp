# 13. Write a program to demonstrate read-only property and write-only property.

using System;

class Student
{
    private int rollNo;
    private string name;

    // 🔹 Read-Only Property
    public int RollNo
    {
        get { return rollNo; }
    }

    // 🔹 Write-Only Property
    public string Name
    {
        set { name = value; }
    }

    // Method to display values
    public void Display()
    {
        Console.WriteLine("Roll No: " + rollNo);
        Console.WriteLine("Name: " + name);
    }

    // Constructor to initialize read-only value
    public Student(int r)
    {
        rollNo = r;
    }
}

class Program
{
    static void Main()
    {
        Student s = new Student(101);

        // Setting write-only property
        s.Name = "John";

        // Accessing read-only property
        Console.WriteLine("Roll No (Read-Only): " + s.RollNo);

        // Display all values
        s.Display();

        Console.ReadLine();
    }
}
