# 6. Design a simple calculator using window form which will have basic operations
# like add, subtract, multiply, divide.

using System;
using System.Windows.Forms;

namespace CalulatorApp
{
	public partial class Form1: Form
	{
		public Form1()
		{
			InitializeComponent();
		}

		private bool GetNumbers(out double num1, out double num2)
		{
			if (!double.TryParse(txtNum1.Text, out num1)) ||
				!double.TryParse(textNum2.Text, out num2)
			{
				MessageBox.Show("Enter valid numbers: ");
				return false;
			}
			return true;
		}

		private void btnAdd_Click(object sender, EventArgs e)
		{
			if (GetNumbers(out double n1, out double n2))
				lblResult.Text = "Result: " + (n1 + n2);
		}

		private void btnSub_Click(object sender, EventArgs e)
		{
			if (GetNumbers(out double n1, out double n2))
				lblResult.Text = "Result: " + (n1 - n2);
		}

		private void btnMul_Click(object sender, EventArgs e)
		{
			if (GetNumbers(out double n1, out double n2))
				lblResult.Text = "Result: " + (n1 * n2);
		}

		private 
	}
}
