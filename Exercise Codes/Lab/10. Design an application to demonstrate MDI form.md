using System;
using System.Windows.Forms;

namespace MDIApp
{
    public partail class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();
        }

        private void openChildToolStripMenuItem_Click(object sender, EventArgs e)
        {
            ChildForm1 child1 = new ChildForm1();
            child1.MdiParent = this;
            child1.Text = "Child Form 1";
            child1.Show();
        }

        private void openChild2ToolStripMenuItem_Click(object sender, EventArgs e)
        {

        }
