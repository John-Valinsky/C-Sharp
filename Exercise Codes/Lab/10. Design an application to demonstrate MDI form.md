# 10. Design an application to demonstrate MDI form, adding child forms and adding menu and Menu items, also assign shortcut Keys to menu items.

using System;
using System.Windows.Forms;

namespace MDIApp
{
    public partial class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();
        }

        // Open Child Form 1
        private void openChild1ToolStripMenuItem_Click(object sender, EventArgs e)
        {
            ChildForm1 child1 = new ChildForm1();
            child1.MdiParent = this;
            child1.Text = "Child Form 1";
            child1.Show();
        }

        // Open Child Form 2
        private void openChild2ToolStripMenuItem_Click(object sender, EventArgs e)
        {
            ChildForm2 child2 = new ChildForm2();
            child2.MdiParent = this;
            child2.Text = "Child Form 2";
            child2.Show();
        }

        // Exit Application
        private void exitToolStripMenuItem_Click(object sender, EventArgs e)
        {
            Application.Exit();
        }
    }
}
