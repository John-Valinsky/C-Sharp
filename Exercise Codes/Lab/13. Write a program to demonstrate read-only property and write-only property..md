# 13. Write a program to demonstrate read-only property and write-only property.

using System;

class Student
{
    private in rollNo;
    private string name;

    public int RollNo
    {
        get { return rollNo; }
    }

    public string Name
    {
        set { name = value; }
    }

    public void Display()
    {
        Console.WriteLine("Roll No: " +  rollNo);
        Console.WriteLine("Name: " + name);
    }

    public Student(int r)
    {
        rollNo = r;
    }
}

class Program
{
    static void Main()
    {
        Student s = new Student(101);

        s.Name = "John";

        Console.WriteLine("Roll No (Read-Only): " + s.RollNo)
    }
}
