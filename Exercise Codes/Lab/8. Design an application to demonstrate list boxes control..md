# 7. Design a window form to demonstrate uses of checkboxes and radio buttons.  

using System;
using System.Text;
using System.Windows.Forms;

namespace SelectionApp
{
    public partial class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();
        }

        private void btnShow_Click(object sender, EventArgs e)
        {
            StringBuilder result = new StringBuilder();

            // CheckBox (Multiple Selection)
            result.Append("Hobbies: ");
            if (chkReading.Checked) result.Append("Reading ");
            if (chkGaming.Checked) result.Append("Gaming ");
            if (chkSports.Checked) result.Append("Sports ");

            // RadioButton (Single Selection)
            result.Append("\nGender: ");
            if (rdoMale.Checked) result.Append("Male");
            else if (rdoFemale.Checked) result.Append("Female");

            lblResult.Text = result.ToString();
        }
    }
}
